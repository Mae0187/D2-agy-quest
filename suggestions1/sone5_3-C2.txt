// debug.ts — 除錯視覺覆蓋層
import type { MissileManager, MissileRuntime } from './manager'; // ← 依實際路徑調整

export interface DebugOptions {
  /** 碰撞半徑（地面像素）。預設讀 m.radius */
  getRadius?: (m: MissileRuntime) => number;
  /** 由 m（通常讀 m.targetId）解析目標螢幕座標；回傳 null 表示無目標。預設不畫線 */
  resolveTarget?: (m: MissileRuntime) => { x: number; y: number } | null;
  getHighWater?: () => number;
  getHits?: () => number;
  /** 邏輯 Tick 數；預設使用內建計數器（需在 loop.onTick 呼叫 noteTick()） */
  getTick?: () => number;
  /** 2:1 等距投影的 Y 壓縮比，預設 0.5 */
  ySquash?: number;
  /** 速度箭頭放大倍數（單位：Tick），預設 6 */
  velocityScale?: number;
  /** 朝向箭頭長度（螢幕像素），預設 24 */
  headingLength?: number;
  hudCorner?: 'tl' | 'tr' | 'bl' | 'br';
}

type Num = { [k: string]: unknown };
const readNum = (o: unknown, key: string): number | undefined => {
  const v = o && (o as Num)[key];
  return typeof v === 'number' ? v : undefined;
};

export class MissileDebugOverlay {
  public enabled = true;
  private ticks = 0;
  private readonly o: Required<Pick<DebugOptions,
    'ySquash' | 'velocityScale' | 'headingLength' | 'hudCorner'>> & DebugOptions;

  constructor(
    private readonly ctx: CanvasRenderingContext2D,
    private readonly manager: MissileManager,
    opt: DebugOptions = {},
  ) {
    this.o = {
      ySquash: 0.5,
      velocityScale: 6,
      headingLength: 24,
      hudCorner: 'tl',
      ...opt,
    };
  }

  toggle(): void { this.enabled = !this.enabled; }

  /** 掛在 MissileLoop.onTick */
  noteTick(): void { this.ticks++; }

  get tick(): number { return this.o.getTick ? this.o.getTick() : this.ticks; }

  /** 世界層 + HUD。若世界座標有相機變換，請分別呼叫 drawWorld / drawHud */
  draw(alpha: number): void {
    this.drawWorld(alpha);
    this.drawHud(alpha);
  }

  /** 碰撞圈、朝向、速度、目標連線（使用目前 ctx 變換，與 renderer 同座標系） */
  drawWorld(alpha: number): void {
    if (!this.enabled || this.manager.liveCount === 0) return;

    const { ctx } = this;
    const a = alpha < 0 ? 0 : alpha > 1 ? 1 : alpha;
    const inv = 1 - a;
    const { ySquash, velocityScale, headingLength } = this.o;
    const pool = this.manager.pool;

    ctx.save();
    try {
      ctx.globalCompositeOperation = 'source-over';
      ctx.globalAlpha = 1;
      ctx.lineWidth = 1;
      ctx.setLineDash([]);

      for (let i = 0; i < pool.length; i++) {
        const m = pool[i];
        if (!m.active) continue;

        // 與 renderer 相同的插值規則；+0.5 讓 1px 線條落在像素中心
        const px = m.teleported ? Math.round(m.x) : Math.round(m.prevX * inv + m.x * a);
        const py = m.teleported ? Math.round(m.y) : Math.round(m.prevY * inv + m.y * a);
        const cx = px + 0.5;
        const cy = py + 0.5;

        // 1) 碰撞圈：地面圓在 2:1 投影下為橢圓
        const r = this.o.getRadius
          ? this.o.getRadius(m)
          : readNum(m, 'radius') ?? 0;
        if (r > 0) {
          ctx.strokeStyle = 'rgba(0,255,255,0.9)';
          ctx.beginPath();
          ctx.ellipse(cx, cy, r, r * ySquash, 0, 0, Math.PI * 2);
          ctx.stroke();
        }

        // 中心十字
        ctx.strokeStyle = 'rgba(255,255,255,0.9)';
        ctx.beginPath();
        ctx.moveTo(cx - 3, cy); ctx.lineTo(cx + 3, cy);
        ctx.moveTo(cx, cy - 3); ctx.lineTo(cx, cy + 3);
        ctx.stroke();

        // 2) 朝向箭頭（phi：地面角 → 螢幕）
        ctx.strokeStyle = 'rgba(255,220,0,0.95)';
        ctx.fillStyle = 'rgba(255,220,0,0.95)';
        arrow(
          ctx, cx, cy,
          cx + Math.cos(m.phi) * headingLength,
          cy + Math.sin(m.phi) * headingLength * ySquash,
        );

        // 3) 速度向量（每 Tick 螢幕位移 × scale）；瞬移幀位移無意義，略過
        if (!m.teleported) {
          const vx = (m.x - m.prevX) * velocityScale;
          const vy = (m.y - m.prevY) * velocityScale;
          if (vx !== 0 || vy !== 0) {
            ctx.strokeStyle = 'rgba(255,0,255,0.95)';
            ctx.fillStyle = 'rgba(255,0,255,0.95)';
            arrow(ctx, cx, cy, cx + vx, cy + vy);
          }
        }

        // 4) 目標鎖定連線
        const t = this.o.resolveTarget ? this.o.resolveTarget(m) : null;
        if (t) {
          ctx.strokeStyle = 'rgba(255,60,60,0.9)';
          ctx.setLineDash([4, 3]);
          ctx.beginPath();
          ctx.moveTo(cx, cy);
          ctx.lineTo(Math.round(t.x) + 0.5, Math.round(t.y) + 0.5);
          ctx.stroke();
          ctx.setLineDash([]);
        }
      }
    } finally {
      ctx.restore(); // 還原 lineDash / 樣式，不污染後續繪製
    }
  }

  /** 角落 HUD。請在「螢幕空間」（無相機變換）呼叫 */
  drawHud(alpha: number): void {
    if (!this.enabled) return;

    const m = this.manager;
    const highWater = this.o.getHighWater
      ? this.o.getHighWater()
      : readNum(m, 'highWater') ?? readNum(m.pool, 'highWater') ?? 0;
    const hits = this.o.getHits
      ? this.o.getHits()
      : readNum((m as unknown as Num).stats, 'hits') ?? 0;

    const lines = [
      `live   ${m.liveCount}`,
      `pool^  ${highWater}`,
      `hits   ${hits}`,
      `tick   ${this.tick}`,
      `alpha  ${alpha.toFixed(2)}`,
    ];

    const { ctx } = this;
    const lh = 14, pad = 6;
    ctx.save();
    try {
      ctx.globalCompositeOperation = 'source-over';
      ctx.globalAlpha = 1;
      ctx.font = '12px monospace';
      ctx.textBaseline = 'top';
      ctx.textAlign = 'left';

      let w = 0;
      for (const l of lines) w = Math.max(w, ctx.measureText(l).width);
      const boxW = Math.ceil(w) + pad * 2;
      const boxH = lines.length * lh + pad * 2;

      const margin = 8;
      const cw = ctx.canvas.width;
      const ch = ctx.canvas.height;
      const c = this.o.hudCorner;
      const bx = c === 'tl' || c === 'bl' ? margin : cw - boxW - margin;
      const by = c === 'tl' || c === 'tr' ? margin : ch - boxH - margin;

      ctx.fillStyle = 'rgba(0,0,0,0.6)';
      ctx.fillRect(bx, by, boxW, boxH);
      ctx.fillStyle = '#7CFC00';
      for (let i = 0; i < lines.length; i++) {
        ctx.fillText(lines[i], bx + pad, by + pad + i * lh);
      }
    } finally {
      ctx.restore();
    }
  }
}

function arrow(
  ctx: CanvasRenderingContext2D,
  x0: number, y0: number, x1: number, y1: number,
  head = 5,
): void {
  ctx.beginPath();
  ctx.moveTo(x0, y0);
  ctx.lineTo(x1, y1);
  ctx.stroke();
  const ang = Math.atan2(y1 - y0, x1 - x0);
  ctx.beginPath();
  ctx.moveTo(x1, y1);
  ctx.lineTo(x1 - head * Math.cos(ang - 0.45), y1 - head * Math.sin(ang - 0.45));
  ctx.lineTo(x1 - head * Math.cos(ang + 0.45), y1 - head * Math.sin(ang + 0.45));
  ctx.closePath();
  ctx.fill();
}

let focused = true;
function makeDoc() {
  return Object.assign(new EventTarget(), { hidden: false, hasFocus: () => focused });
}

function makeRaf() {
  let next = 1;
  const q = new Map<number, (t: number) => void>();
  return {
    raf: (cb: (t: number) => void) => { const id = next++; q.set(id, cb); return id; },
    caf: (id: number) => { q.delete(id); },
    get pending() { return q.size; },
    /** 以指定時間戳觸發目前排程中的那一個 rAF 回呼 */
    fire(t: number) {
      const first = q.entries().next().value;
      if (!first) throw new Error('no pending rAF');
      q.delete(first[0]);
      first[1](t);
    },
  };
}

}

## 附錄：F03 祝福之槌（阿基米德螺旋）擴展指引

**前提聲明**：我沒有 3-B 的原始碼，無法確認私有狀態陣列的實際欄位名與行為原語註冊方式。下列以「每個投射物有一段可讀寫的數值槽（Float64Array 區段或 `number[]`）」為前提，**純運算核心與銜接點分開寫**。核心可獨立測試，只有 §3 的銜接需要依 3-B 實際介面改寫。螺旋參數是可調的設計值，**不是**從遊戲資料表擷取的原始數值。

### 1. 數學設計

螺旋定義在**地面空間**（壓縮前），再投影到螢幕。若直接在螢幕座標畫螺旋，地面軌跡會是橢圓而非圓。

- 極座標：`r = r0 + b·Δθ`，Δθ 為累計角度
- 地面座標：`(gx, gy) = (r·cosθ, r·sinθ)`
- 螢幕座標：`(cx + gx, cy + gy·ySquash)`，`ySquash = 0.5`
- **用絕對位置而非速度積分**：每 Tick 從 `(θ, r, 圓心)` 重算位置，無累積誤差，也不會和 manager 的位移積分疊加。
- **等弧長速度**：定角速度的阿基米德螺旋，外圈切向速度會越來越快。要讓槌子看起來等速，取弧長微元 `ds = √(r²+b²)·dθ`，即 `dθ = speed / hypot(r, b)`，並以 `maxDTheta` 限制 `r≈0` 時的暴衝。
- **朝向 `phi`**：取切線方向 `(b·cosθ − dir·r·sinθ, b·sinθ + dir·r·cosθ)` 的 `atan2`，`dir=±1` 表示順/逆時針。

### 2. 純運算核心（spiral.ts）

```ts
// spiral.ts — 與 3-B 介面無關的純函式，可直接單元測試
export const SP = { THETA: 0, R: 1, CX: 2, CY: 3, DIR: 4, SIZE: 5 } as const;

export interface SpiralParams {
  r0: number;         // 起始半徑（地面 px）
  b: number;          // 每弧度徑向增量（地面 px/rad），須 > 0
  speed: number;      // 沿弧線速度（地面 px/Tick）
  rMax: number;       // 半徑達此值即結束
  maxDTheta: number;  // 單 Tick 最大角增量（rad）
  ySquash: number;    // 0.5
}

export interface SpiralOut { x: number; y: number; phi: number; done: boolean }

/** 生成時呼叫一次。cx, cy 為圓心（通常 = 施法者/生成點，螢幕座標） */
export function spiralInit(
  s: ArrayLike<number> & { [i: number]: number }, o: number,
  cx: number, cy: number, theta0: number, dir: 1 | -1, p: SpiralParams,
): void {
  s[o + SP.THETA] = theta0;
  s[o + SP.R] = p.r0;
  s[o + SP.CX] = cx;
  s[o + SP.CY] = cy;
  s[o + SP.DIR] = dir;
}

/** 每 Tick 呼叫一次；結果寫入 out（無配置） */
export function spiralStep(
  s: ArrayLike<number> & { [i: number]: number }, o: number,
  p: SpiralParams, out: SpiralOut,
): void {
  const r = s[o + SP.R];
  const dir = s[o + SP.DIR];
  const dTheta = Math.min(p.speed / Math.hypot(r, p.b), p.maxDTheta);

  const theta = s[o + SP.THETA] + dir * dTheta;
  const nr = r + p.b * dTheta;
  s[o + SP.THETA] = theta;
  s[o + SP.R] = nr;

  const c = Math.cos(theta), sn = Math.sin(theta);
  out.x = s[o + SP.CX] + nr * c;
  out.y = s[o + SP.CY] + nr * sn * p.ySquash;

  out.phi = Math.atan2(p.b * sn + dir * nr * c, p.b * c - dir * nr * sn);
  out.done = nr >= p.rMax;
}
```

### 3. 銜接 3-B（需依實際介面調整）

先對照 3-B 的三件事再動手：

1. **狀態槽怎麼配置**：若每個投射物有固定長度的私有槽，確認長度 ≥ `SP.SIZE`（5）。若是共用大陣列加偏移，`o` 就是該投射物的起點。若槽數不足，先擴 `SIZE` 與物件池的重置邏輯，**回收時必須清零**，否則舊螺旋狀態會殘留給下一發。
2. **原語簽名與註冊表**：新增一個與既有原語同形的 `spiralOut`，在 spec 中指定它與 `SpiralParams`。典型形狀如下（`[...]` 部分依 3-B 調整）：

```ts
// 在 3-B 的原語表中新增（示意）
const out: SpiralOut = { x: 0, y: 0, phi: 0, done: false };

function onSpawnSpiral(m: MissileRuntime, p: SpiralParams, dir: 1 | -1) {
  spiralInit(/* m 的私有狀態槽 */, /* 偏移 */ 0, m.x, m.y, m.phi, dir, p);
}

function stepSpiral(m: MissileRuntime, p: SpiralParams) {
  spiralStep(/* m 的私有狀態槽 */, 0, p, out);
  m.x = out.x;
  m.y = out.y;
  m.phi = out.phi;
  if (out.done) { /* 呼叫 3-B 既有的「到期/死亡」路徑 */ }
}
```

3. **`prevX/prevY` 由誰設**：若 `manager.step()` 在行為執行前已做 `prev = cur`，原語內**不要再碰** `prevX/prevY`，否則插值失效。若原語自己負責，才在寫入新位置前 `m.prevX = m.x`。圓心若要跟隨移動中的施法者，改為每 Tick 傳入圓心，不存在槽內，並注意此時圓心位移會讓槌子位置突變，必要時改用「圓心增量」。

另外注意：

- **座標單位**：若 3-B 的 `x, y` 是定點數（例如 8.8），在 `spiralStep` 前後轉換，核心內部維持浮點。
- **確定性**：若有回放/網路同步需求，`Math.cos/sin` 的跨平台差異不可忽略，改用 3-A 的整數角度表（如 `indexToPhi` 對應的查表）。
- **碰撞掃掠**：外圈切向速度在 `speed` 固定時恆定，不會穿透。若調高 `speed`，請確認單 Tick 位移小於碰撞半徑。
- **渲染**：槌子的自旋動畫用 `speed` 256 進位推進即可，與 `phi` 無關。`phi` 的用途是讓除錯層的朝向箭頭驗證切線方向（箭頭應始終與螺旋相切）。

### 4. 螺旋核心的單元測試（不依賴 3-B）

```ts
// spiral.test.ts
import { describe, it, expect } from 'vitest';
import { SP, spiralInit, spiralStep, type SpiralParams, type SpiralOut } from './spiral';

const P: SpiralParams = { r0: 8, b: 6, speed: 4, rMax: 200, maxDTheta: 0.5, ySquash: 0.5 };

function run(ticks: number, dir: 1 | -1 = 1) {
  const s = new Float64Array(SP.SIZE);
  spiralInit(s, 0, 400, 300, 0, dir, P);
  const out: SpiralOut = { x: 0, y: 0, phi: 0, done: false };
  const pts: { x: number; y: number; r: number }[] = [];
  for (let i = 0; i < ticks; i++) {
    spiralStep(s, 0, P, out);
    pts.push({ x: out.x, y: out.y, r: s[SP.R] });
  }
  return { pts, out };
}

describe('F03 阿基米德螺旋核心', () => {
  it('半徑單調遞增', () => {
    const { pts } = run(100);
    for (let i = 1; i < pts.length; i++) expect(pts[i].r).toBeGreaterThan(pts[i - 1].r);
  });

  it('地面圓經 2:1 投影：((x-cx)/r)² + ((y-cy)/(r·0.5))² = 1', () => {
    for (const p of run(80).pts) {
      const u = (p.x - 400) / p.r;
      const v = (p.y - 300) / (p.r * P.ySquash);
      expect(u * u + v * v).toBeCloseTo(1, 9);
    }
  });

  it('等弧長：地面空間相鄰兩點距離 ≈ speed（r 足夠大、未被 maxDTheta 限制時）', () => {
    const { pts } = run(120);
    for (let i = 60; i < 119; i++) {
      const gx = pts[i + 1].x - pts[i].x;
      const gy = (pts[i + 1].y - pts[i].y) / P.ySquash; // 還原到地面空間
      expect(Math.hypot(gx, gy)).toBeGreaterThan(P.speed * 0.85);
      expect(Math.hypot(gx, gy)).toBeLessThan(P.speed * 1.15);
    }
  });

  it('dir=-1 與 dir=1 以圓心鏡像（x 相同、y 對圓心反向）', () => {
    const a = run(30, 1).pts, b = run(30, -1).pts;
    for (let i = 0; i < 30; i++) {
      expect(b[i].x).toBeCloseTo(a[i].x, 9);
      expect(b[i].y - 300).toBeCloseTo(-(a[i].y - 300), 9);
    }
  });

  it('達到 rMax 時 done=true', () => {
    expect(run(5).out.done).toBe(false);
    expect(run(2000).out.done).toBe(true);
  });
});
```

這份測試只驗證數學核心。接上 3-B 後，建議再加一個整合測試：用 manager 跑 N 個 Tick，確認 `m.x/m.y` 與 `spiralStep` 單獨算出的軌跡一致，並確認 `prevX/prevY` 恰為上一 Tick 的位置。這樣可以抓到「prev 被重複設定」或「槽位未在回收時清零」這兩類最常見的接線錯誤。
